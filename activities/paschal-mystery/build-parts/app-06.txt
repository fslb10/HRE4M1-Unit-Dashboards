 if(n===6){if(BELIEFS.some((_,i)=>!RATINGS.includes(state.finalRatings[i])))return fail('Review and record a final judgement for every belief.');if(!enough('revision',25,'change in thinking')||!enough('unity',40,'one unified saving action'))return;if(!EVENT_NAMES.every(e=>state.forms.unity.toLowerCase().includes(e.toLowerCase())))return fail('Include all four events by name in your explanation of the unified saving action.');if(!state.reviewed)return fail('Confirm that the team has reviewed its argument.');if(![1,2,3,4].every(i=>state.done.includes(i)))return fail('Recover all four seals before attempting the final vault.');if(state.forms.code!==state.digits.join(''))return fail('The digits do not match. Read the four recovered seals in order, I to IV.');state.mission.remaining=missionRemaining();state.mission.running=false;state.completedAt=new Date().toISOString();}
 state.done.push(n);if(n===0)state.diagnosisLockedAt=new Date().toISOString();if(n===6)chime();else if(n>=1&&n<=4)chime();writeState();nextStage();toast(n>=1&&n<=4?`Seal ${['I','II','III','IV'][n-1]} recovered · digit ${state.digits[n-1]}`:n===6?'Archive restored. Prepare your group defence.':'Record saved.');
}

function action(a){
 if(a==='home'){view='home';showSetup=false;render();window.scrollTo({top:0,behavior:'instant'});if(view==='game')$('#roomTitle')?.focus({preventScroll:true});}
 else if(a==='teacher'){view='teacher';render();window.scrollTo({top:0,behavior:'instant'});if(view==='game')$('#roomTitle')?.focus({preventScroll:true});}
 else if(a==='choose'||a==='setup'){showSetup=true;render();$('#setup')?.scrollIntoView({behavior:'smooth',block:'start'});$('#teamName')?.focus({preventScroll:true});}
 else if(a==='resume'){if(state){view='game';render();window.scrollTo({top:0,behavior:'instant'});if(view==='game')$('#roomTitle')?.focus({preventScroll:true});}}
 else if(a==='complete')complete();
 else if(a==='check-evidence'){evidenceChecked=true;SCENARIOS[state.event].evidence.forEach(c=>opened[c.id]=true);const bad=SCENARIOS[state.event].evidence.filter(c=>state.evidence[c.id]!==c.kind);feedback[state.stage]={type:bad.length?'info':'success',message:bad.length?`${bad.length} record${bad.length===1?' needs':'s need'} another look. Review the feedback below.`:'All six classifications are supported. Now explain why your two strongest records belong together.'};renderRoom(false);}
 else if(a==='hint'){if(!state.hintsUsed.includes(state.stage)&&state.hints>0){state.hints--;state.hintsUsed.push(state.stage);writeState();}renderRoom(false);$('.hint')?.scrollIntoView({behavior:'smooth',block:'center'});}
 else if(a==='mission-pause'){if(!state.mission.duration){toast('This investigation is untimed.');return;}if(state.mission.running){state.mission.remaining=missionRemaining();state.mission.running=false;}else{state.mission.end=Date.now()+state.mission.remaining*1000;state.mission.running=true;}writeState();render();}
 else if(a==='suggest-challenge'){const recipient=state.forms.recipient;if(!EVENT_NAMES.includes(recipient)||recipient===state.event){toast('Choose another event first.');return;}setForm('challenge',SCENARIOS[recipient].challenge);renderRoom(false);}
 else if(a==='share-challenge'){shareChallenge();}
 else if(a==='copy-question'){if(!state.forms.challenge){toast('Write your question first.');return;}copy(`${state.team} → ${state.forms.recipient||'another team'}\n${state.forms.challenge}`,'Question copied.');}
 else if(a==='accept-challenge'){state.incoming=pendingChallenge;writeState();render();toast('Challenge added to your notebook.');}
 else if(a==='timer-toggle'){const t=getTimer();if(t.running){t.remaining=timerRemaining(t);t.running=false;}else{if(t.remaining<=0)t.remaining=phaseDurations(t)[t.phase];t.end=Date.now()+t.remaining*1000;t.running=true;}persistTimer();updateTimers();}
 else if(a==='timer-reset'){const t=getTimer();t.running=false;t.remaining=phaseDurations(t)[t.phase];persistTimer();updateTimers();}
 else if(a==='timer-next'){const t=getTimer();t.phase=(t.phase+1)%3;t.running=false;t.remaining=phaseDurations(t)[t.phase];persistTimer();refreshTimer();}
 else if(a==='fullscreen'){const e=$('#defenceTimer');if(document.fullscreenElement)document.exitFullscreen?.();else if(e?.requestFullscreen)e.requestFullscreen().catch(()=>toast('Fullscreen is unavailable in this browser.'));else toast('Use your browser’s fullscreen or zoom controls.');}
 else if(a==='finish'){if(!state.done.includes(7))state.done.push(7);state.defendedAt=new Date().toISOString();writeState();render();toast('Defence recorded as complete. Export your report.');}
 else if(a==='export'){download(report(),'paschal-'+safeName(state.team)+'-report.txt','text/plain;charset=utf-8');toast('Report downloaded. Submit it as your teacher directs.');}
 else if(a==='backup'){writeState();download(JSON.stringify(state,null,2),'paschal-'+safeName(state.team)+'-backup.json','application/json');}
 else if(a==='import'){importBackup();}
 else if(a==='print'){const p=$('#printReport');p.textContent=report();window.print();}
 else if(a==='notebook'){openModal('Team notebook',`<div class="actions"><button class="btn primary" id="modalReport">Download report</button><button class="btn" id="modalBackup">Save progress backup</button></div><pre>${esc(report())}</pre><label class="field"><span class="field-title">Additional notes / live challenge response</span><textarea id="extraNotes" maxlength="6000">${f('extraNotes')}</textarea></label><p class="fine muted">Notes save locally as you type.</p>`);$('#modalReport').onclick=()=>action('export');$('#modalBackup').onclick=()=>action('backup');$('#extraNotes').oninput=e=>setForm('extraNotes',e.target.value);}
