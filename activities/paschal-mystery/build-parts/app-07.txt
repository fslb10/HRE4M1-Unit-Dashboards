 $$('[data-timer-phase]').forEach(b=>b.addEventListener('click',()=>{const t=getTimer();t.phase=Number(b.dataset.timerPhase);t.running=false;t.remaining=phaseDurations(t)[t.phase];persistTimer();refreshTimer();}));
 $('[data-timer-auto]')?.addEventListener('change',e=>{getTimer().auto=e.target.checked;persistTimer();updateTimers();});
 $('[data-timer-extended]')?.addEventListener('change',e=>{const t=getTimer();t.extended=e.target.checked;t.running=false;t.remaining=phaseDurations(t)[t.phase];persistTimer();refreshTimer();});

}

function formatTime(s){s=Math.max(0,Math.ceil(Number(s)||0));return Math.floor(s/60).toString().padStart(2,'0')+':'+(s%60).toString().padStart(2,'0');}
function timerRemaining(t){return t.running?Math.max(0,(t.end-Date.now())/1000):t.remaining;}
function missionRemaining(){const m=state?.mission;return m?m.running?Math.max(0,(m.end-Date.now())/1000):m.remaining:0;}
function updateTimers(){
 const m=state?.mission;
 if(m){$$('[data-mission-clock]').forEach(e=>e.textContent=!m.duration?'Untimed':missionRemaining()<=0?'Time guide complete':formatTime(missionRemaining())+(m.running?'':' / paused'));}
 const t=getTimer();if(!t)return;
 const left=timerRemaining(t),dur=phaseDurations(t);
 $$('[data-timer-clock]').forEach(e=>e.textContent=formatTime(left));
 $$('[data-timer-label]').forEach(e=>e.textContent=['Make your case','Ask one focused question','Answer the challenge'][t.phase]);
 $$('[data-timer-progress]').forEach(e=>e.style.width=clamp(left/dur[t.phase]*100,0,100)+'%');
 $$('[data-timer-button]').forEach(e=>e.textContent=(t.running?'Pause':'Start')+' timer');
 $$('[data-timer-phase]').forEach((e,i)=>{e.setAttribute('aria-pressed',String(i===t.phase));e.textContent=['Defence','Cross-examination','Response'][i]+' · '+dur[i]+'s';});
 $$('[data-timer-status]').forEach(e=>e.textContent=t.running?'Timer running.':left===0?'Phase complete. Advance when the next speaker is ready.':'Ready. Press Start.');
 if(t.running&&left<=0){t.running=false;t.remaining=0;chime();$('#announcer').textContent=['Defence','Cross-examination','Response'][t.phase]+' time is complete.';if(t.auto&&t.phase<2){t.phase++;t.remaining=dur[t.phase];t.end=Date.now()+t.remaining*1000;t.running=true;}persistTimer();}
}
